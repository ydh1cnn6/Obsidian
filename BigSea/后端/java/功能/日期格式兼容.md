/**  
 * str - >LocalDate * @param dateStr   日期字符串格式  
 * @param unit 作用：支持指定日期单位用的
 * 比如：生产碰到yyyyMM，LocalDate转换不了yyyyMM格式，就单独处理
 * @return 默认值：null  
 */
public LocalDate getLocadeDateByStrAndUnit(String dateStr, ChronoUnit unit) {  
    LocalDate parse = null;  
    if (StringUtils.isBlank(dateStr) || unit == null) {  
        return null;  
    }  
    //  
    String[] split = "yyyyMMdd,yyyy-MM-dd,yyyy-MM-dd HH:mm:ss,yyyy/mm/dd,yyyy/mm/dd HH:mm:ss".split(",");  
    for (String s : split) {  
        try {  
            parse = LocalDate.parse(s, DateTimeFormatter.ofPattern(s));  
        }catch (Exception e) {  
            e.printStackTrace();  
        }  
    }  
    //只有年月的日期，提供补足日的支持（日默认给01）  
    if (ChronoUnit.MONTHS.equals(unit)) {  
        if (parse==null){  
            String[] splitYearMonth = "yyyyMM,yyyy-MM,yyyy/MM".split(",");  
            for (String s : splitYearMonth) {  
                try {  
                    parse = YearMonth.parse(s, DateTimeFormatter.ofPattern(s)).atDay(1);      
}catch (Exception e) {  
                    e.printStackTrace();  
                }  
            }  
        }  
    }  
    return parse;  
}
