const expInput = c.querySelector("#exp");
const resultDiv = c.querySelector("#result");
const historyDiv = c.querySelector("#history_panel");
const historyHead = c.querySelector("#history_head");

// 折叠切换
let historyOpen = false;
historyHead.addEventListener("click", () => {
  historyOpen = !historyOpen;
  historyDiv.style.maxHeight = historyOpen ? "120px" : "0";
  historyHead.innerHTML = historyOpen ? "历史记录 ▲" : "历史记录 ▼";
});

function renderHistory() {
  historyDiv.innerHTML = history.map(h => `<div>${h}</div>`).join("");
}

// 按钮输入
c.querySelectorAll("button[btn]").forEach(btn => {
  btn.addEventListener("click", () => {
    const val = btn.getAttribute("btn");
    if (val === "C") {
      expInput.value = "";
    } else if (val === "CE") {
      expInput.value = expInput.value.replace(/\d+$/, "");
    } else if (val === "backspace") {
      expInput.value = expInput.value.slice(0, -1);
    } else if (val === "^2") {
      expInput.value += "^2";
    } else if (val === "sqrt") {
      expInput.value += "Math.sqrt(";
    } else if (val === "1/") {
      expInput.value += "1/";
    } else {
      expInput.value += val;
    }
    expInput.focus();
  });
});

// 计算函数
function calculate() {
  let expr = expInput.value.trim()
    .replace(/×/g, "*")
    .replace(/÷/g, "/")
    .replace(/\^2/g, "**2")
    // ✅ 关键：把 ^ 自动换成 ** 次方
    .replace(/\^/g, "**")
    .replace(/¹⁄ₓ/g, "1/")
    .replace(/（/g, "(")
    .replace(/）/g, ")");

  if (!expr) return;

  try {
    const res = Function("'use strict';return (" + expr + ")")();
    const line = `${expr} = ${res}`;
    history.unshift(line);
    saveHistory();
    resultDiv.innerText = res;
    resultDiv.style.display = "block";
    renderHistory();
  } catch (e) {
    resultDiv.innerText = "表达式错误";
    resultDiv.style.display = "block";
  }
}

// 点击 = 计算
c.querySelector("#calc_eq").addEventListener("click", calculate);

// 回车自动计算
expInput.addEventListener("keydown", e => {
  if (e.key === "Enter") calculate();
});

renderHistory();
