const adapter = app.vault.adapter;
const snippetsDir = `${app.vault.configDir}/snippets`;

// 1. 后台自动创建缺失的 CSS 文件
async function initCSS() {
    if (!(await adapter.exists(snippetsDir))) {
        await adapter.mkdir(snippetsDir);
    }
    for (const style of styles) {
        const path = `${snippetsDir}/${style.id}.css`;
        if (!(await adapter.exists(path))) {
            await adapter.write(path, cssContents[style.id]);
        }
    }
}
initCSS();

// 2. 渲染按钮界面
const container = dv.container;
container.empty();
const btnContainer = container.createEl('div', { 
    attr: { style: 'display: flex; gap: 8px; flex-wrap: wrap; margin: 15px 0;' } 
});

// 创建按钮的辅助函数
function createBtn(text, onClick, isActive = false) {
    const btn = btnContainer.createEl('button', { text: text });
    btn.style.padding = '6px 12px';
    btn.style.borderRadius = '6px';
    btn.style.cursor = 'pointer';
    btn.style.border = '1px solid var(--background-modifier-border)';
    btn.style.backgroundColor = isActive ? 'var(--interactive-accent)' : 'var(--background-secondary)';
    btn.style.color = isActive ? 'var(--text-on-accent)' : 'var(--text-normal)';
    btn.style.transition = 'all 0.2s ease';
    btn.addEventListener('click', async () => {
        await onClick();
        renderButtons();
    });
}

// 3. 互斥切换逻辑
async function switchStyle(targetId) {
    // 先全部关闭
    for (const style of styles) {
        app.customCss.setCssEnabledStatus(style.id, false);
    }
    // 开启选中的
    if (targetId !== 'original') {
        app.customCss.setCssEnabledStatus(targetId, true);
        new Notice(`🟢 已切换至：${styles.find(s => s.id === targetId).name}`);
    } else {
        new Notice("🔴 已恢复原始分割线");
    }
}

// 4. 渲染按钮
function renderButtons() {
    btnContainer.empty();
    for (const style of styles) {
        const isActive = app.customCss.enabledSnippets.has(style.id);
        createBtn(style.name, () => switchStyle(style.id), isActive);
    }
    const isOriginal = styles.every(s => !app.customCss.enabledSnippets.has(s.id));
    createBtn("🔴 恢复原始", () => switchStyle('original'), isOriginal);
}

// 初始化渲染
renderButtons();
